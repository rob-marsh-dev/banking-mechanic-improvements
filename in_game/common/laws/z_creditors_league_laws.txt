creditors_league_payment_law = {
	type = international_organization
	law_category = finance
	potential = {
		international_organization_type = international_organization_type:creditors_league
	}
	allow = { always = yes }
	requires_vote = yes
	custom_tags = { forbids_no_policy }

	# Tier 1: Leader subsidises members — small gold payment from bank to each member
	cl_subsidy_policy = {
		payments_repealed = { cl_small_levy_payment cl_large_levy_payment cl_full_levy_payment }
		payments_implemented = { cl_member_subsidy_payment }
		country_modifier = {
			potential_trigger = {
				NOT = { is_leader_of_international_organization = scope:recipient }
			}
			bank_interest = -0.005
		}
		wants_this_policy_bias = {
			if = {
				limit = { is_leader_of_international_organization = scope:recipient }
				add = { desc = "CL_LEADER_PAYS_SUBSIDY" value = -10 }
			}
			else = {
				add = { desc = "CL_MEMBER_RECEIVES_SUBSIDY" value = 40 }
			}
		}
		wants_keep_policy = {
			if = {
				limit = { is_leader_of_international_organization = scope:recipient }
				add = { desc = "CL_LEADER_WANTS_CHANGE" value = -20 }
			}
			else = {
				add = { desc = "CL_MEMBER_WANTS_KEEP" value = 30 }
			}
		}
	}

	# Tier 2: Free association — no payments in either direction
	cl_free_association_policy = {
		payments_repealed = { cl_member_subsidy_payment }
		wants_this_policy_bias = {
			add = { desc = "CL_NEUTRAL_TERMS" value = 5 }
		}
		wants_keep_policy = {
			add = { desc = "CL_NEUTRAL_TERMS" value = 5 }
		}
	}

	# Tier 3: Small levy — members pay 2% of income to leader
	cl_small_levy_policy = {
		payments_repealed = { cl_member_subsidy_payment }
		payments_implemented = { cl_small_levy_payment }
		country_modifier = {
			potential_trigger = {
				is_leader_of_international_organization = scope:recipient
			}
			monthly_loan_capacity_investment = 2.5
		}
		wants_this_policy_bias = {
			if = {
				limit = { is_leader_of_international_organization = scope:recipient }
				add = { desc = "CL_LEADER_GAINS_LEVY" value = 30 }
			}
			else = {
				add = { desc = "CL_MEMBER_PAYS_LEVY" value = -20 }
				add = {
					desc = "CL_MEMBER_INCOME_WEIGHT"
					value = monthly_income_trade_and_tax
					multiply = -0.1
				}
			}
		}
		wants_keep_policy = {
			if = {
				limit = { is_leader_of_international_organization = scope:recipient }
				add = { desc = "CL_LEADER_KEEPS_LEVY" value = 30 }
			}
			else = {
				add = { desc = "CL_MEMBER_RESISTS_LEVY" value = -15 }
			}
		}
	}

	# Tier 4: Large levy — members pay 5% of income to leader
	cl_large_levy_policy = {
		payments_repealed = { cl_small_levy_payment }
		payments_implemented = { cl_large_levy_payment }
		allow = {
			international_organization_has_policy = policy:cl_small_levy_policy
		}
		country_modifier = {
			potential_trigger = {
				NOT = { is_leader_of_international_organization = scope:recipient }
			}
			bank_interest = 0.005
		}
		country_modifier = {
			potential_trigger = {
				is_leader_of_international_organization = scope:recipient
			}
			monthly_loan_capacity_investment = 5.0
			global_merchant_capacity_modifier = 0.05
		}
		wants_this_policy_bias = {
			if = {
				limit = { is_leader_of_international_organization = scope:recipient }
				add = { desc = "CL_LEADER_GAINS_LEVY" value = 50 }
			}
			else = {
				add = { desc = "CL_MEMBER_PAYS_LEVY" value = -40 }
				add = {
					desc = "CL_MEMBER_INCOME_WEIGHT"
					value = monthly_income_trade_and_tax
					multiply = -0.2
				}
			}
		}
		wants_keep_policy = {
			if = {
				limit = { is_leader_of_international_organization = scope:recipient }
				add = { desc = "CL_LEADER_KEEPS_LEVY" value = 50 }
			}
			else = {
				add = { desc = "CL_MEMBER_RESISTS_LEVY" value = -30 }
			}
		}
	}

	# Tier 5: Full levy — members pay 10% of income to leader
	cl_full_levy_policy = {
		payments_repealed = { cl_large_levy_payment }
		payments_implemented = { cl_full_levy_payment }
		allow = {
			international_organization_has_policy = policy:cl_large_levy_policy
		}
		country_modifier = {
			potential_trigger = {
				NOT = { is_leader_of_international_organization = scope:recipient }
			}
			bank_interest = 0.01
			selling_efficiency = -0.01
		}
		country_modifier = {
			potential_trigger = {
				is_leader_of_international_organization = scope:recipient
			}
			monthly_loan_capacity_investment = 10.0
			global_merchant_capacity_modifier = 0.1
		}
		wants_this_policy_bias = {
			if = {
				limit = { is_leader_of_international_organization = scope:recipient }
				add = { desc = "CL_LEADER_GAINS_LEVY" value = 80 }
			}
			else = {
				add = { desc = "CL_MEMBER_PAYS_LEVY" value = -80 }
				add = {
					desc = "CL_MEMBER_INCOME_WEIGHT"
					value = monthly_income_trade_and_tax
					multiply = -0.5
				}
			}
		}
		wants_keep_policy = {
			if = {
				limit = { is_leader_of_international_organization = scope:recipient }
				add = { desc = "CL_LEADER_KEEPS_LEVY" value = 80 }
			}
			else = {
				add = { desc = "CL_MEMBER_RESISTS_LEVY" value = -60 }
			}
		}
