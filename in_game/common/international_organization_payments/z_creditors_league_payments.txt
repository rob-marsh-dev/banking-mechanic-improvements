# Tier 1 — leader subsidises members (1% of member income paid out proportionally)
cl_member_subsidy_payment = {
	get_payer_list = {
		leader_country ?= {
			add_to_local_variable_list = {
				name = payers
				target = this
			}
		}
	}
	get_payee_list = {
		every_international_organization_member = {
			limit = { NOT = { is_leader_of_international_organization = root } }
			add_to_local_variable_list = {
				name = payees
				target = this
			}
		}
	}
	price = cl_payment_base
	uses_maintenance = no
	price_multiplier = {
		every_international_organization_member = {
			limit = { NOT = { is_leader_of_international_organization = root } }
			add = {
				desc = "CL_PRICE_TOTAL_MEMBER_INCOME"
				value = monthly_income_trade_and_tax
			}
		}
		multiply = {
			desc = "CL_SUBSIDY_RATE"
			value = 0.01
		}
	}
	proportion_for_payer = {
		add = { desc = "cl_payment_all" value = 1 }
	}
	proportion_for_payee = {
		add = {
			desc = "CL_PRICE_YOUR_INCOME_SHARE"
			value = monthly_income_trade_and_tax
		}
	}
}
