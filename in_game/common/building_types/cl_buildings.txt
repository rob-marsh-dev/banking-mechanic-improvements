# Creditor's League economic buildings.
#
# Built BY the league leader (a banking country) IN member territory, as a gold sink that
# also boosts the member's local economy. Gating:
#   country_potential : builder must be a banking country leading a Creditor's League
#   allow             : the target location's owner must be a member of THE BUILDER'S league,
#                       and that league must have cl_con_build_policy passed
# (allow scopes: root = location, scope:actor = the country building it)

cl_trade_house = {
	is_foreign = yes
	pop_type = burghers
	category = trade_category

	max_levels = 1
	employment_size = 0.2
	pop_size_created = 0.1

	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes

	country_potential = {
		has_reform = government_reform:banking_country
		is_leader_of_international_organization_of_type = { type = creditors_league }
	}

	allow = {
		owner = {
			any_international_organizations_member_of = {
				international_organization_type = international_organization_type:creditors_league
				leader_country = scope:actor
				international_organization_has_policy = policy:cl_con_build_policy
			}
		}
	}

	build_time = bank_office_build_time

	modifier = {
		local_merchant_capacity = 2
		local_market_access = 0.1	}

	unique_production_methods = {
		cl_trade_house_maintenance = {
			paper = 0.2
			jewelry = .1
			category = building_maintenance
		}
	}

	construction_demand = bank_construction
}
