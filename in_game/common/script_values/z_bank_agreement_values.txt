# Numbers for the phase 3 agreements. All placeholders until a balance pass.
#
# Which of scope:actor / scope:recipient is the host differs between contexts in
# scripted relations, so the sums below pick whichever side is NOT the bank
# (same trick as lending_rights_signing_bonus).

### Marriage Alliance ###
# Dowry the bank pays the host: a year of the host's trade and tax income.
bank_marriage_alliance_dowry = {
	value = 0
	if = {
		limit = {
			exists = scope:actor
			exists = scope:recipient
			scope:actor = { has_reform = government_reform:banking_country }
		}
		add = scope:recipient.monthly_income_trade_and_tax
	}
	else_if = {
		limit = {
			exists = scope:actor
			exists = scope:recipient
		}
		add = scope:actor.monthly_income_trade_and_tax
	}
	multiply = 12
	min = 0
}

### Standing Retainer ###
# Monthly fee the bank pays, as a share of the HOST's trade and tax income.
bank_standing_retainer_share = 0.1

# Gold the bank must have in hand to sign: a year of the fee.
bank_standing_retainer_reserve = {
	value = 0
	if = {
		limit = {
			exists = scope:actor
			exists = scope:recipient
			scope:actor = { has_reform = government_reform:banking_country }
		}
		add = scope:recipient.monthly_income_trade_and_tax
	}
	else_if = {
		limit = {
			exists = scope:actor
			exists = scope:recipient
		}
		add = scope:actor.monthly_income_trade_and_tax
	}
	multiply = bank_standing_retainer_share
	multiply = 12
	min = 0
}
