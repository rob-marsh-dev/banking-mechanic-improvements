# root = subject, scope:overlord = new overlord, scope:subject_type = subject type
on_subject_created = {
	on_actions = { cl_on_subject_created }
}

cl_on_subject_created = {
	effect = {
		if = {
			limit = {
				scope:overlord = {
					any_international_organizations_member_of = {
						international_organization_type = international_organization_type:creditors_league
						international_organization_has_policy = policy:cl_sub_forbidden_policy
						NOT = { leader_country ?= scope:overlord }
					}
				}
			}
			scope:overlord = { cancel_subject = root }
		}
	}
}

# root = country (borrower), scope:target = loan, scope:owner = lender (if existent)
# When a Creditor's League member takes a loan, retag the new loan so the league LEADER
# is the creditor (a country loan) instead of the borrower's own estates. The borrower
# already received the principal from the estate loan; change_loan_owner only reassigns
# who collects interest/repayment, so there is no double-gold.
#
# UNVERIFIED: whether change_loan_owner works on an estate loan. It no-ops on bonds
# (see project-bond-buying). If it no-ops here too, fall back to cancel_loan + give_loan
# with explicit gold handling.
on_loan_taken = {
	on_actions = { cl_on_loan_taken }
}

cl_on_loan_taken = {
	effect = {
		if = {
			limit = {
				# Borrower is a Creditor's League member whose league has a leader other than itself.
				any_international_organizations_member_of = {
					international_organization_type = international_organization_type:creditors_league
					exists = leader_country
					NOT = { leader_country = root }
				}
				# Only retag an ownerless, non-bond loan (i.e. the estate loan just taken).
				scope:target = {
					is_bond = no
					NOT = { exists = owner }
				}
			}
			random_international_organizations_member_of = {
				limit = {
					international_organization_type = international_organization_type:creditors_league
					exists = leader_country
					NOT = { leader_country = root }
				}
				leader_country = { save_temporary_scope_as = bank_leader }
			}
			# Permanent (not temporary) so it survives into the notification event.
			save_scope_as = cl_loan_taker
			scope:target = {
				change_loan_owner = scope:bank_leader
			}
			# The bank actually fronts the principal, so it isn't free money for the league.
			scope:bank_leader = {
				add_gold = {
					value = scope:target.loan_amount
					multiply = -1
				}
				# Notify a human bank that a member just drew on its treasury.
				# AI leaders skip the popup (a country_event to AI only auto-resolves silently).
				if = {
					limit = { is_ai = no }
					set_variable = { name = cl_last_loan_amount value = scope:target.loan_amount }
					trigger_event_non_silently = { id = banking_cl.1 }
				}
			}
		}
	}
}
