# Marriage Alliance: the bank pays a dowry to marry one of its own into a
# country's ruling family.
#
# What it does:
#   - the bank pays the host a dowry (bank_marriage_alliance_dowry)
#   - a NEW character is created in the host's court. They belong to the bank
#     ruler's dynasty (same family name) but are not an existing family member.
#   - that character is wed to the host's heir, else its ruler, else any unwed
#     adult of the ruling dynasty. If nobody is free they simply join the court.
#   - while the agreement lasts both sides get opinion and trust with each other
#     (opinion_/trust_ giving_/receiving_bank_marriage_alliance in
#     common/biases/z_banking_biases.txt; the engine applies these by name).
#
# For the later throne takeover / IO: the host stores the spouse in the variable
# bank_marriage_spouse, and the spouse stores the bank in bank_marriage_house.
#
# One way: the HOST gives the relation, the BANK receives it. Only the bank can
# initiate (request). Lasts until cancelled or broken by war.
# Built on the Lending Rights relation; see z_lending_rights.txt for the notes on
# why roles are resolved by the banking reform instead of by scope name.
bank_marriage_alliance = {
	type = diplomacy
	relation_type = oneway
	break_on_war = yes
	relation_type_for_ai = friendly
	diplomatic_cost_request = bank_marriage_alliance_cost
	category = {
		offer = CATEGORY_BANKING_ACTIONS
		request = CATEGORY_BANKING_ACTIONS
		cancellation = CATEGORY_BANKING_ACTIONS
		break = CATEGORY_BANKING_ACTIONS
	}

	# Only banks can buy it, and not from another bank.
	visible = {
		OR = {
			NOT = { exists = scope:recipient }
			scope:recipient = { has_reform = government_reform:banking_country }
		}
		OR = {
			NOT = { exists = scope:actor }
			scope:actor = {
				NOT = { has_reform = government_reform:banking_country }
			}
		}
	}

	offer_visible = { always = no }

	request_enabled = {
		exists = scope:actor
		exists = scope:recipient
		# Both houses need a ruler with a family name to marry into / from.
		scope:actor = {
			has_ruler = yes
			ruler ?= { has_dynasty = yes }
		}
		scope:recipient = {
			has_ruler = yes
			ruler ?= { has_dynasty = yes }
		}
		OR = {
			AND = {
				scope:actor = { has_reform = government_reform:banking_country }
				# One such agreement per host.
				scope:recipient = {
					NOT = { giving_scripted_relation_of_type = relation_type:bank_marriage_alliance }
				}
				scope:actor.gold >= bank_marriage_alliance_dowry
			}
			AND = {
				scope:recipient = { has_reform = government_reform:banking_country }
				scope:actor = {
					NOT = { giving_scripted_relation_of_type = relation_type:bank_marriage_alliance }
				}
				scope:recipient.gold >= bank_marriage_alliance_dowry
			}
		}
	}

	will_expire_trigger = {
		scope:second = {
			NOT = { has_reform = government_reform:banking_country }
		}
	}

	request_effect = {
		# Work out which side is the bank.
		if = {
			limit = { scope:actor = { has_reform = government_reform:banking_country } }
			scope:actor = { save_scope_as = bma_bank }
			scope:recipient = { save_scope_as = bma_host }
		}
		else = {
			scope:recipient = { save_scope_as = bma_bank }
			scope:actor = { save_scope_as = bma_host }
		}
		# Fix the amount once so both sides see the same number.
		save_scope_value_as = { name = bma_payment value = bank_marriage_alliance_dowry }
		scope:bma_bank = {
			add_gold = {
				value = scope:bma_payment
				multiply = -1
			}
		}
		scope:bma_host = {
			add_gold = scope:bma_payment
		}
		custom_tooltip = bank_marriage_alliance_spouse_tooltip

		hidden_effect = {
			# Who in the host's ruling family is free to wed?
			scope:bma_host = {
				if = {
					limit = {
						heir ?= {
							is_adult = yes
							has_available_marriage_slot = yes
						}
					}
					heir = { save_scope_as = bma_partner }
				}
				else_if = {
					limit = {
						ruler ?= {
							is_adult = yes
							has_available_marriage_slot = yes
						}
					}
					ruler = { save_scope_as = bma_partner }
				}
				else = {
					ruler ?= {
						dynasty ?= {
							random_character_in_dynasty = {
								limit = {
									owner ?= scope:bma_host
									is_adult = yes
									has_available_marriage_slot = yes
								}
								save_scope_as = bma_partner
							}
						}
					}
				}

				# The new member of the banking family, of the opposite sex to the
				# partner (a daughter of the house if nobody is free).
				if = {
					limit = {
						exists = scope:bma_partner
						scope:bma_partner = { is_female = yes }
					}
					create_character = {
						dynasty = scope:bma_bank.ruler.dynasty
						culture = scope:bma_bank.ruler.culture
						religion = scope:bma_bank.ruler.religion
						estate = estate_type:nobles_estate
						female = no
						min_age = 18
						max_age = 30
						save_scope_as = bma_spouse
					}
				}
				else = {
					create_character = {
						dynasty = scope:bma_bank.ruler.dynasty
						culture = scope:bma_bank.ruler.culture
						religion = scope:bma_bank.ruler.religion
						estate = estate_type:nobles_estate
						female = yes
						min_age = 16
						max_age = 25
						save_scope_as = bma_spouse
					}
				}
			}

			if = {
				limit = { exists = scope:bma_spouse }
				if = {
					limit = { exists = scope:bma_partner }
					scope:bma_partner = { marry_character_ignore_blocks = scope:bma_spouse }
				}
				scope:bma_spouse = {
					set_variable = { name = bank_marriage_house value = scope:bma_bank }
				}
				scope:bma_host = {
					set_variable = { name = bank_marriage_spouse value = scope:bma_spouse }
				}
			}
		}
	}

	cancel_effect = {
		scope:actor ?= { remove_variable = bank_marriage_spouse }
		scope:recipient ?= { remove_variable = bank_marriage_spouse }
	}

	break_effect = {
		scope:actor ?= { remove_variable = bank_marriage_spouse }
		scope:recipient ?= { remove_variable = bank_marriage_spouse }
	}

	wants_to_give_diplo_chance = {
		base = -20
		actor_is_rival = -100
		trust_in_actor = 0.5
		positive_opinion = 0.2
		negative_opinion = -0.5
	}

	# The host (scope:actor) deciding whether to accept the match.
	wants_to_give = {
		add = {
			desc = "BASE"
			value = -50
		}
		# The dowry is the sweetener.
		add = {
			desc = "bank_marriage_alliance_dowry_reason"
			value = 50
		}
		add = {
			desc = "THEIR_OPINION_TOOLTIP"
			value = "scope:actor.opinion(scope:recipient)"
			if = {
				limit = { "scope:actor.opinion(scope:recipient)" > 0 }
				divide = 5
			}
		}
		add = {
			desc = "THEIR_TRUST_TOOLTIP"
			value = "scope:actor.trust(scope:recipient)"
			divide = 2
		}
		if = {
			limit = {
				scope:actor = { is_rival_of = scope:recipient }
			}
			add = {
				desc = "game_concept_rival"
				value = -200
			}
		}
	}
}
