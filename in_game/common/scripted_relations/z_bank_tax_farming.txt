# Tax Farming: the bank pays a country a signing sum, and in return collects a
# share of that country's tax income every month.
#
# One way: the HOST gives the relation, the BANK receives it. Only the bank can
# initiate (request). Lasts until cancelled or broken by war.
# Built on the Lending Rights relation; see z_lending_rights.txt for the notes on
# why roles are resolved by the banking reform instead of by scope name.
# In gold_to_second: scope:first = host (giver), scope:second = bank (receiver).
bank_tax_farming = {
	type = diplomacy
	relation_type = oneway
	break_on_war = yes
	relation_type_for_ai = friendly
	# The bank's cut: a tenth of the host's tax income (income excluding trade, so
	# it never overlaps with Trade Factorage).
	gold_to_second = {
		value = scope:first.monthly_income_trade_and_tax
		subtract = scope:first.monthly_trade_income
		multiply = bank_tax_farming_share
		min = 0
	}
	diplomatic_cost_request = bank_tax_farming_cost
	category = {
		offer = CATEGORY_BANKING_ACTIONS
		request = CATEGORY_BANKING_ACTIONS
		cancellation = CATEGORY_BANKING_ACTIONS
		break = CATEGORY_BANKING_ACTIONS
	}

	# Only banks can buy it, and not from another bank.
	visible = {
		OR = {
			NOT = { exists = scope:recipient }
			scope:recipient = { has_reform = government_reform:banking_country }
		}
		OR = {
			NOT = { exists = scope:actor }
			scope:actor = {
				NOT = { has_reform = government_reform:banking_country }
			}
		}
	}

	offer_visible = { always = no }

	# Roles are resolved by the banking reform rather than by scope name, since
	# actor/recipient are not consistent across contexts (see header).
	request_enabled = {
		exists = scope:actor
		exists = scope:recipient
		OR = {
			AND = {
				scope:actor = { has_reform = government_reform:banking_country }
				# One such agreement per host.
				scope:recipient = {
					NOT = { giving_scripted_relation_of_type = relation_type:bank_tax_farming }
				}
				scope:actor.gold >= bank_tax_farming_signing_bonus
			}
			AND = {
				scope:recipient = { has_reform = government_reform:banking_country }
				scope:actor = {
					NOT = { giving_scripted_relation_of_type = relation_type:bank_tax_farming }
				}
				scope:recipient.gold >= bank_tax_farming_signing_bonus
			}
		}
	}

	will_expire_trigger = {
		scope:second = {
			NOT = { has_reform = government_reform:banking_country }
		}
	}

	request_effect = {
		# Work out which side is the bank, then pay the borrower for signing.
		if = {
			limit = { scope:actor = { has_reform = government_reform:banking_country } }
			scope:actor = { save_scope_as = btf_bank }
			scope:recipient = { save_scope_as = btf_borrower }
		}
		else = {
			scope:recipient = { save_scope_as = btf_bank }
			scope:actor = { save_scope_as = btf_borrower }
		}
		# Fix the amount once so both sides see the same number.
		save_scope_value_as = { name = btf_payment value = bank_tax_farming_signing_bonus }
		scope:btf_bank = {
			add_gold = {
				value = scope:btf_payment
				multiply = -1
			}
		}
		scope:btf_borrower = {
			add_gold = scope:btf_payment
		}
	}

	wants_to_give_diplo_chance = {
		base = -10
		actor_is_rival = -100
		trust_in_actor = 0.5
		positive_opinion = 0.1
		negative_opinion = -0.5
	}

	# The borrower (scope:actor) deciding whether to sell its lending rights.
	wants_to_give = {
		add = {
			desc = "BASE"
			value = -50
		}
		# The signing bonus is the sweetener.
		add = {
			desc = "bank_tax_farming_signing_bonus_reason"
			value = 50
		}
		add = {
			desc = "THEIR_OPINION_TOOLTIP"
			value = "scope:actor.opinion(scope:recipient)"
			if = {
				limit = { "scope:actor.opinion(scope:recipient)" > 0 }
				divide = 5
			}
		}
		add = {
			desc = "THEIR_TRUST_TOOLTIP"
			value = "scope:actor.trust(scope:recipient)"
			divide = 2
		}
		if = {
			limit = {
				scope:actor = { is_rival_of = scope:recipient }
			}
			add = {
				desc = "game_concept_rival"
				value = -200
			}
		}
	}
}
