# Mint Contract: the bank runs a country's mint. The host mints more cleanly
# (giving_bank_mint_contract static modifier: more minting income, more minting
# before inflation) and the bank takes a monthly cut that grows with how hard the
# host is running its mint.
#
# There is no script value for minting income itself, so the cut is
#   host income x minting slider (0..1) x bank_mint_contract_share.
# A host that mints nothing pays nothing.
#
# One way: the HOST gives the relation, the BANK receives it. Only the bank can
# initiate (request). Lasts until cancelled or broken by war.
# Built on the Lending Rights relation; see z_lending_rights.txt for the notes on
# why roles are resolved by the banking reform instead of by scope name.
# In gold_to_second: scope:first = host (giver), scope:second = bank (receiver).
bank_mint_contract = {
	type = diplomacy
	relation_type = oneway
	break_on_war = yes
	relation_type_for_ai = friendly
	gold_to_second = {
		value = scope:first.monthly_income_trade_and_tax
		multiply = scope:first.slider_minting_value
		multiply = bank_mint_contract_share
		min = 0
	}
	giving_modifier_scale = {
		add = {
			desc = "BASE"
			value = 1
		}
	}
	diplomatic_cost_request = bank_mint_contract_cost
	category = {
		offer = CATEGORY_BANKING_ACTIONS
		request = CATEGORY_BANKING_ACTIONS
		cancellation = CATEGORY_BANKING_ACTIONS
		break = CATEGORY_BANKING_ACTIONS
	}

	# Only banks can buy it, and not from another bank.
	visible = {
		OR = {
			NOT = { exists = scope:recipient }
			scope:recipient = { has_reform = government_reform:banking_country }
		}
		OR = {
			NOT = { exists = scope:actor }
			scope:actor = {
				NOT = { has_reform = government_reform:banking_country }
			}
		}
	}

	offer_visible = { always = no }

	request_enabled = {
		exists = scope:actor
		exists = scope:recipient
		OR = {
			AND = {
				scope:actor = { has_reform = government_reform:banking_country }
				# One such agreement per host.
				scope:recipient = {
					NOT = { giving_scripted_relation_of_type = relation_type:bank_mint_contract }
				}
				scope:actor.gold >= bank_mint_contract_signing_bonus
			}
			AND = {
				scope:recipient = { has_reform = government_reform:banking_country }
				scope:actor = {
					NOT = { giving_scripted_relation_of_type = relation_type:bank_mint_contract }
				}
				scope:recipient.gold >= bank_mint_contract_signing_bonus
			}
		}
	}

	will_expire_trigger = {
		scope:second = {
			NOT = { has_reform = government_reform:banking_country }
		}
	}

	request_effect = {
		# Work out which side is the bank, then pay the host for signing.
		if = {
			limit = { scope:actor = { has_reform = government_reform:banking_country } }
			scope:actor = { save_scope_as = bmc_bank }
			scope:recipient = { save_scope_as = bmc_host }
		}
		else = {
			scope:recipient = { save_scope_as = bmc_bank }
			scope:actor = { save_scope_as = bmc_host }
		}
		# Fix the amount once so both sides see the same number.
		save_scope_value_as = { name = bmc_payment value = bank_mint_contract_signing_bonus }
		scope:bmc_bank = {
			add_gold = {
				value = scope:bmc_payment
				multiply = -1
			}
		}
		scope:bmc_host = {
			add_gold = scope:bmc_payment
		}
	}

	wants_to_give_diplo_chance = {
		base = -10
		actor_is_rival = -100
		trust_in_actor = 0.5
		positive_opinion = 0.1
		negative_opinion = -0.5
	}

	# The host (scope:actor) deciding whether to hand over its mint.
	wants_to_give = {
		add = {
			desc = "BASE"
			value = -50
		}
		# The signing bonus is the sweetener.
		add = {
			desc = "bank_mint_contract_signing_bonus_reason"
			value = 50
		}
		add = {
			desc = "THEIR_OPINION_TOOLTIP"
			value = "scope:actor.opinion(scope:recipient)"
			if = {
				limit = { "scope:actor.opinion(scope:recipient)" > 0 }
				divide = 5
			}
		}
		add = {
			desc = "THEIR_TRUST_TOOLTIP"
			value = "scope:actor.trust(scope:recipient)"
			divide = 2
		}
		if = {
			limit = {
				scope:actor = { is_rival_of = scope:recipient }
			}
			add = {
				desc = "game_concept_rival"
				value = -200
			}
		}
	}
}
