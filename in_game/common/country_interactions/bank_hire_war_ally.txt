# Hire War Ally: a bank's one-off alliance. The bank pays a country gold to join
# one specific war it is leading. Nothing persists afterwards: no relation, no
# alliance slot, no obligation in the next war.
# Cloned from vanilla ask_join_war_for_favors, with the favor cost replaced by a
# gold price paid to the hired country, and the requirement that the country is
# already callable (allied) removed.
bank_hire_war_ally = {
	type = diplomacy
	category = CATEGORY_BANKING_ACTIONS

	price = price:bank_hire_war_ally_price
	payee = scope:recipient
	
	potential = {
		scope:actor = {
			has_reform = government_reform:banking_country
			at_war = yes
			any_current_war = {
				not = { is_in_war = root }
				or = {
					attacker_leader = scope:actor
					defender_leader = scope:actor
				}
			}
		}
	}

	allow = {
	}
	
	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		show_why_not_enabled = yes
		none_available_msg_key = "no_valid_countries"
		default_sort = will
		column = {
			data = country_diplomacy_info_military
		}
		visible = {
		}
        enabled = {
			scope:actor = {
				any_current_war = {
					not = { is_in_war = root }
					trigger_if = {
						limit = {
							attacker_leader = scope:actor
						}
						always = yes
					}
					trigger_else_if = {
						limit = {
							defender_leader = scope:actor
						}
						always = yes
					}
					trigger_else = {
						always = no
					}
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = war
		target_flag = war
		interaction_source_list = {
			scope:actor = {
				every_current_war = {
					add_to_list = source
				}
			}
		}
		cache_interaction_source_list = yes
		name = "choose_war_to_join"
		default_sort = select_target_acceptance_frame
		column = {
			data = name
		}
		column = {
			data = select_target_acceptance_frame
		}
		visible = {
		}
        enabled = {
			not = { is_in_war = scope:recipient }

			trigger_if = {
				limit = {
					is_civil_war_for = scope:actor
					scope:actor = { is_subject_or_below_of = scope:recipient }
				}
				war_length > MONTHS_BEFORE_OVERLORD_CAN_JOIN_CIVIL_WAR
			}

			trigger_if = {
				limit = {
					attacker_leader = scope:actor
				}
				can_join_as_attacker = scope:recipient
				NOT = {
					defender_leader = {
						is_subject_or_below_of = scope:recipient
					}
				}
				scope:recipient = {
					NOT = {
						has_truce_with = root.defender_leader
					}
				}
			}
			trigger_else_if = {
				limit = {
					defender_leader = scope:actor
				}
				can_join_as_defender = scope:recipient
				NOT = {
					attacker_leader = {
						is_subject_or_below_of = scope:recipient
					}
				}
				scope:recipient = {
					NOT = {
						has_truce_with = root.attacker_leader
					}
				}
			}
			trigger_else = {
				always = no
			}
		}
	}

	ai_will_do = {
		add = {
			desc = "TARGET_ACCEPTANCE"
			value = "scope:actor.country_interaction_acceptance(country_interaction:bank_hire_war_ally|scope:recipient)"
		}
	}

	diplo_chance = {
		base = 40
		levy_availability = -100
		low_manpower = -20
		negative_stability = -0.2
		war_exhaustion = -2
		in_debt = -5
		recipient_civil_war = -100
		recipient_at_war = -100
		recipient_occupied_beseiged_locations = -100
		disloyal_subject = -100
		opinion = 0.3
		trust_in_actor = 0.5
		lacks_border = -10
		border_distance = -0.1
	}

	accept = {
		if = {
			limit = { exists = scope:war }
			if = {
				limit = { scope:war.attacker_leader = scope:actor }
				if = {
					limit = { scope:war = { any_defender = { in_union_with = root } } }
					subtract = {
						desc = "DIPLOREASON_WOULD_BREAK_UNION"
						value = 1000
					}
				}
				if = {
					limit = { is_allied_with = { target = scope:war.defender_leader } }
					subtract = {
						desc = "DIPLOREASON_ALLIED_TO_ENEMY"
						value = 20
					}
				}
				subtract = {
					desc = "DIPLOREASON_TARGET_OPINION"
					value = "opinion(scope:war.defender_leader)"
					multiply = 0.5
				}
				subtract = {
					desc = "DIPLOREASON_BORDER_DISTANCE"
					value = "border_distance_to(scope:war.defender_leader)"
					multiply = 0.1
				}
			}
			else = {
				if = {
					limit = { scope:war = { any_attacker = { in_union_with = root } } }
					subtract = {
						desc = "DIPLOREASON_WOULD_BREAK_UNION"
						value = 1000
					}
				}
				if = {
					limit = { is_allied_with = { target = scope:war.attacker_leader } }
					subtract = {
						desc = "DIPLOREASON_ALLIED_TO_ENEMY"
						value = 20
					}
				}
				subtract = {
					desc = "DIPLOREASON_TARGET_OPINION"
					value = "opinion(scope:war.attacker_leader)"
					multiply = 0.5
				}
				subtract = {
					desc = "DIPLOREASON_BORDER_DISTANCE"
					value = "border_distance_to(scope:war.attacker_leader)"
					multiply = 0.1
				}
			}
		}
	}
	
	effect = {
        scope:war = {
			if = {
				limit = {
					attacker_leader = scope:actor
				}
				scope:recipient = {
					join_war_as_attacker = {
						war = prev
						call_in_subjects = yes
					}
				}
			}
			else_if = {
				limit = {
					defender_leader = scope:actor
				}
				scope:recipient = {
					join_war_as_defender = {
						war = prev
						call_in_subjects = yes
					}
				}
			}
		}
	}
}
