hostile_takeover_into_league = {
	type = diplomacy
	category = CATEGORY_HOSTILE_ACTIONS

	potential = {
		scope:actor = {
			has_reform = government_reform:banking_country
			is_member_of_international_organization_of_type = { type = creditors_league }
		}
		scope:recipient = {
			NOT = { is_member_of_international_organization_of_type = { type = creditors_league } }
			NOT = { has_reform = government_reform:banking_country }
		}
	}

	allow = {
		scope:recipient = {
			any_loan = {
				count >= 5
				exists = owner
				owner = scope:actor
				months_left > 20
			}
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			NOT = { is_member_of_international_organization_of_type = { type = creditors_league } }
			NOT = { has_reform = government_reform:banking_country }
		}
		enabled = {
			any_loan = {
				count >= 5
				exists = owner
				owner = scope:actor
				months_left > 20
			}
		}
	}

	effect = {
		scope:actor = {
			every_international_organizations_member_of = {
				limit = {
					international_organization_type = international_organization_type:creditors_league
				}
				add_country_to_international_organization = scope:recipient
			}
		}
	}

	ai_will_do = {
		value = 0
		if = {
			limit = {
				scope:recipient = {
					monthly_income_trade_and_tax > 10
					any_loan = {
						count >= 5
						exists = owner
						owner = scope:actor
					}
				}
			}
			add = 40
		}
	}
}
