take_over_all_loans = {
	category = CATEGORY_ECONOMY_ACTIONS
	type = diplomacy

	potential = {
		scope:actor = {
			has_reform = government_reform:banking_country
		}
	}

	allow = {
		scope:actor = {
			has_reform = government_reform:banking_country
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			num_loans > 0
		}
		enabled = {
			any_loan = {
				NOT = { exists = owner }
				months_left > 40
				loan_interest > scope:actor.modifier:bank_interest
			}
			scope:actor.gold > banking_ownerless_loan_principal
		}
	}

	effect = {
		custom_description = {
			text = take_over_all_loans_count_tooltip
			value = scope:recipient.num_loans
		}
		custom_description = {
			text = take_over_all_loans_debt_tooltip
			value = scope:recipient.total_debt
		}
		custom_description = {
			text = take_over_all_loans_cost_tooltip
			value = scope:recipient.banking_ownerless_loan_principal
		}
		scope:recipient = {
			every_loan = {
				limit = {
					NOT = { exists = owner }
					months_left > 40
					loan_interest > scope:actor.modifier:bank_interest
				}
				custom_description = {
					text = pay_off_their_debt
					value = bank_loan_takeover_cost
					scope:actor = {
						add_gold = {
							value = prev.bank_loan_takeover_cost
							multiply = -1
						}
					}
					scope:recipient = {
						distribute_gold_to_banking_estates = prev.bank_loan_takeover_cost
					}
				}
				change_loan_owner = scope:actor
			}
		}
	}

	ai_will_do = {
		add = -20
		add = scope:recipient.country_tax_base
	}
}
